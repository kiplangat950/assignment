const calculationHistory = []; function addToHistory(num1, num2, operator, result) { const logEntry = { operands: [num1, num2], operator: operator, result: result }; calculationHistory.push(logEntry); } function add(a, b) { const result = a + b; addToHistory(a, b, '+', result); return result; } function subtract(a, b) { const result = a - b; addToHistory(a, b, '-', result); return result; } function multiply(a, b) { const result = a * b; addToHistory(a, b, '*', result); return result; } function divide(a, b) { if (b === 0) { const errorMsg = "Error: Division by zero"; addToHistory(a, b, '/', errorMsg); return errorMsg; } const result = a / b; addToHistory(a, b, '/', result); return result; } function displayHistory() { if (calculationHistory.length === 0) { console.log("You have no stored calculations."); return; } console.log("\n Calculation History Log "); calculationHistory.forEach((entry, index) => { const [num1, num2] = entry.operands; console.log(`[${index + 1}] ${num1} ${entry.operator} ${num2} = ${entry.result}`); }); console.log("===============================\n"); } displayHistory(); add(10, 5); subtract(20, 8); multiply(4, 2.5); divide(100, 4); divide(7, 0); displayHistory();
